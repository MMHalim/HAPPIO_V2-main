SELECT
  LEFT(interaction_id, POSITION('-' IN interaction_id) - 1) as Conversation_ID,
  id
FROM
  "Freshworks"
WHERE 
  "Channel" = 'T2 Product Support'
  AND ("assigned_qa" = '' OR "assigned_qa" IS NULL) 
  AND ("HAPPIO_Status" = '' OR "HAPPIO_Status" IS NULL)
  AND "Year" >= '{{txtyyyy.text}}' 
  AND "Year" <= '{{txtyyyyTo.text}}'
  AND "Month" >= '{{txtmm.text}}' 
  AND "Month" <= '{{txtmmTo.text}}'
  AND "Day" >= '{{txtdd.text}}' 
  AND "Day" <= '{{txtddTo.text}}'
